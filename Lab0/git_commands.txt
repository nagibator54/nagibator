git clone скачать репозиториий 
git status показывает какие файлы ты изменил
git add добовляет файлы в очередь на сохранение 
git commit сохраняет все добавленые фацлы с коментом 
git push отправляет сохраненые файлы 
git pull скачивает свежие изменения 
git branch показывает все ветки или воздает новую ветку 
git checkout переключается тебя на другую ветку 
git merge склеивает изменения из одной ветки в другую 
git log показывает историю всем коммитов
